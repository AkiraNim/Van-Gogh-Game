#Handle Quest Signals
#NOTES: there are two types of quests — talk and collect types.
# talk_quests demand the player to talk to a specified npc target.
# collect_quests demand the player to collect the specified item_target defined by a global variable.

[signal arg="quest_talk_hit:NPC"] #Completes a talk_quest type. 
#Receives one String parameter — the name of the NPC target.
#When triggered, it completes any accepted talk quest that has 'talk_target' matching the NPC name.

[signal arg="quest_accept:coletar_item:collect"] #Accepts a collect_quest type and creates it in quests_active.
#Receives two String parameters: the first is the quest_name, and the second is the quest_type.
#Sets status = "accepted" and registers the quest as active.

[signal arg="quest_set_req:coletar_item:item_id:1"] #Defines the requirements of a quest.
#Receives three parameters: quest_name, item_id, and the quantity of that item required for completion.

[signal arg="quest_add:coletar_item:item_id:{count_item}"] #Adds progress to a quest based on a variable.
#Receives three parameters: quest_name, progress_id (the item), and a variable representing the quantity to add.
#If total progress meets or exceeds the requirement, the quest auto-completes.

[signal arg="npc_next:NPC"] #Makes the specified NPC advance to its next timeline.
#Receives one String parameter: the name of the NPC.
#Internally calls the NPC’s method 'avancar_timeline()' to proceed to the next dialog state.
